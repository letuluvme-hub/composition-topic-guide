/* 从题目推概念：命中多个档就取并集。全都不命中就返回 null ——
   宁可说"我没把握这道题该调哪些概念"，也不要瞎猜一通然后装作很准。 */
function conceptsForTopic(topic){
  var t=norm(topic).replace(/[《》「」""'']/g,''), i, k;
  var got={}, hitProfiles=0, list=[];
  for(i=0;i<TOPIC_PROFILES.length;i++){
    var p=TOPIC_PROFILES[i], matched=false;
    for(k=0;k<p.k.length;k++) if(t.indexOf(p.k[k])>=0){ matched=true; break; }
    if(!matched) continue;
    /* 「美」太泛，只有在没有更具体的档命中时才用 */
    if(p.weak && hitProfiles) continue;
    hitProfiles++;
    for(k=0;k<p.c.length;k++) if(!got[p.c.charAt(k)]){ got[p.c.charAt(k)]=1; list.push(p.c.charAt(k)); }
  }
  if(!list.length) return null;
  return {c:list, profiles:hitProfiles};
}
function profileNote(topic){
  var t=norm(topic).replace(/[《》「」""'']/g,''), i, k;
  for(i=0;i<TOPIC_PROFILES.length;i++){
    var p=TOPIC_PROFILES[i];
    for(k=0;k<p.k.length;k++) if(t.indexOf(p.k[k])>=0 && !p.weak) return p.note;
  }
  return '';
}
